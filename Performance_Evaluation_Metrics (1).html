department_summary = (
    df.groupby("Department")
    .agg(
        Interns=("Intern ID", "count"),
        Avg_Task_Completion=("Task Completion Rate (%)", "mean"),
        Avg_Project_Quality=("Project Quality KPI (%)", "mean"),
        Avg_Mentor_Feedback=("Mentor Feedback KPI (%)", "mean"),
        Avg_Attendance=("Attendance KPI (%)", "mean"),
        Avg_Performance=("Overall Performance Score", "mean")
    )
    .round(2)
    .reset_index()
)
